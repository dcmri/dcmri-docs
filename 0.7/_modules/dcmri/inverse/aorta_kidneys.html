class InverseAortaKidneys(Module):

    configs = configs
    defaults = defaults
    quantities = {
        'E_li': {'init': 0.0, 'bounds': (0.0, 1.0), 'name': 'extraction fraction in liver', 'unit': '', 'group': 'physiological', 'dicom_key': None, 'osipi_key': None},
    }

    _all_inputs = {'Tb', 'Db', 'D_p_lk', 'D_p_rk', 'T_vc', 'D_vc', 'nb', 'pfree', 'Nz', 'S0_lk', 'dose_1', 'F_u_lk', 'PA', 'F_u_rk', 'H', 'h_u_rk', 'NSR_rk', 'iStrig_ao', 'S_rk', 'TD', 'B1corr_lk', 'TA', 'T_hl', 'v_b', 'iStrig_rk', 'S0_rk', 'bdel', 'NSR_lk', 'T_b_or', 'S0_ao', 'S_ao', 'agent', 'rate', 'TF', 'iStrig_lk', 'tS_lk', 'R1_ki', 'T_gu', 'rate_1', 'vol_lk', 'T_u_lk', 'D_hl', 'TP', 'tS_ao', 'TR', 'B1corr_ao', 'T_lka', 'T_rka', 'dose_2', 'DRPF', 'S_lk', 'tS_rk', 'T_e_or', 'dose_tolerance', 'NSR_ao', 'v_ki', 'dose', 'dt', 'v_p_lk', 'rate_2', 'Nph', 'TE', 'F_u', 'vol_rk', 'TE1', 'TE2', 'E_or', 'FF_lk', 'R1_c', 'iz', 'FA', 'E_li', 'PSw', 'v_p_rk', 'R1_b', 'Nk0', 'FF_rk', 'weight', 'vol_ao', 'B1corr_rk', 'T_u_rk', 'h_u_lk', 'CO', 'fCO_ki', 'R1_u', 'v_u', 'SA', 'me', 'field_strength', 'v_c'}
    _all_outputs = {'psdev', 'pcov', 'popt', 'pars', 'loss'}

    def __init__(self, imap:dict=None, omap:dict=None, **config):
        self.set_config(config)
        self.forward = ForwardModel(**self.config)
        self.map_io(imap, omap)

    def _predict(self, p):
        pred = self.forward(p)
        time = tuple([p[f'tS_{roi}'].reshape(-1) for roi in ROIS])
        return tuple([pred[f'S_{roi}'][:, :, :len(time[i])].reshape(-1) for i, roi in enumerate(ROIS)])

    def _preproc(self, p):
        # Derive simulation timings
        p[f'tstart'] = min([p[f'tS_{roi}'][0] for roi in ROIS])
        tmax = max([p[f'tS_{roi}'][-1] for roi in ROIS])
        ts = p[f'tS_ao'][1] - p[f'tS_ao'][0]
        p[f'tacq'] = tmax - p[f'tstart'] + 2 * ts

        # Reshape signal if needed
        for roi in ROIS:
            if p[f'S_{roi}'].ndim == 1:
                p[f'S_{roi}'] = p[f'S_{roi}'].reshape(1, 1, -1)

        # Set calibration signal
        if self.config['calibrate']:
            for roi in ROIS:
                p[f"Scal_{roi}"] = p[f"S_{roi}"][..., :p['nb']]
                p[f'iScal_{roi}'] = np.arange(p['nb'])

        # Trigger signals
        if self.config['trigger']:
            for roi in ROIS:
                if p[f'iStrig_{roi}'] is not None:
                    p[f'tS_{roi}'] = p[f'tS_{roi}'][p[f'iStrig_{roi}']]  
                    p[f'S_{roi}'] = p[f'S_{roi}'][:, :, p[f'iStrig_{roi}']] 

    def pfree(self):
        inputs = self.forward.mapped_inputs()
        pfree = {p for p in inputs if Qty(p).group=='physiological'}
        pfree |= {'BAT'}
        if not self.config['calibrate']:
            pfree |= {'S0_ao', 'S0_lk', 'S0_rk'}
        pfree -= {'H', 'E_li', 'v_ki'}
        return {p: Qty(p).bounds for p in pfree}

    def _pder(self, p: dict, popt:dict):
        data = p | popt
        p = {k: v for k, v in data.items() if Qty(k).group in ['physiological', 'whole-body']}
        p = dpars_kidney(p, kinetics=self.config['kidneys'])
        p |= {k: v for k, v in data.items() if Qty(k).name == 'BAT'}
        return p | popt

    def _call(self, p: dict=None, **kwargs) -> dict:
        self._preproc(p) 

        # Estimate BAT if needed
        if self.config['estimate_bat'] == 'custom':
            p['BAT'] = estimate_bat(p['tS_ao'], p['S_ao'])
        elif self.config['estimate_bat'] == 'ttp':
            p['BAT'] = p['tS_ao'][np.argmax(p['S_ao'])]

        if p['pfree'] is None:
            p['pfree'] = self.pfree() 
        else:
            p['pfree'] = self.map_data(p['pfree'], all=False, include={'BAT'})
        p['pfree'] = update_bounds(p['pfree'], value=p, all=False)

        # Compute inverse
        signal = tuple([p[f'S_{roi}'].reshape(-1) for roi in ROIS])
        output = train_brute(self._predict, signal, p, p['pfree'], **kwargs)
        output['pars'] = self._pder(p, output['popt'])

        return output

    def inputs(self) -> set:
        inputs = self.forward.mapped_inputs()
        inputs |= {'pfree'}
        for roi in ROIS:
            inputs |= {f'tS_{roi}', f'S_{roi}'}
        if self.config['calibrate']:
            inputs |= {'nb'}
            for roi in ROIS:
                inputs -= {f'Scal_{roi}', f'iScal_{roi}'}
        if self.config['estimate_bat'] != 'none':
            inputs -= {'BAT'}
        inputs -= {f'tstart', f'tacq'}
        return inputs  
    
    def outputs(self):
        return {'popt', 'psdev', 'pcov', 'pars', 'loss'}

    def init_data(self):
        p = super().init_data()
        p['E_li'] = 0
        return p
    
    def test_data(self, data: dict=None): 
        p = self.init_data()
        p |= self.forward.test_data()

        pred = self.forward(p)
        p |= {
            'nb': 5,
            'pfree': self.forward.filter_data({'CO': (10, 300), 'BAT': (-60, 60)}),
        }
        for roi in ROIS:        
            p |= {
                f'tS_{roi}': pred[f'tS_{roi}'],
                f'S_{roi}': pred[f'S_{roi}'], 
            }
               
        return self.input_data(p, data)


    def plot(self, data: dict, result:dict=None, xlim:list=None, fname:str=None, show=True):
        if result is None:
            result = self.__call__(data)
        p = self.map_data(data | result['popt'])
        
        # Not an input parameter but required by forward model
        if 'BAT' in result['popt']:
            p['BAT'] = result['popt']['BAT']

        self._preproc(p)
        pred = self.forward(p)

        if xlim is None: 
            xlim = [pred['tR'][0], pred['tR'][-1]]
        xlim = np.array(xlim) / 60
        
        fig, axes = plt.subplots(3, 2, figsize=(10, 8))
        fig.subplots_adjust(wspace=0.3, hspace=0.4)
        ((ax1, ax2), (ax3, ax4), (ax5, ax6)) = axes
        
        # Plot signals
        ax1.set_title('MRI Signal Prediction')
        def plot_data(t, s, ti, si, ax, clr):
            ax.set(xlabel='Time (min)', ylabel='Signal (a.u.)', xlim=xlim)
            for i in range(si.shape[0]):
                for j in range(si.shape[1]):
                    ax.plot(ti / 60, si[i, j, :], marker='o', color=clr[0], alpha=0.5, label='Data')
                    ax.plot(t / 60, s[i, j, :], linestyle='-', color=clr[1], linewidth=3, label='Prediction')                
            ax.legend()

        plot_data(pred['tS_ao'], pred['S_ao'], pred['tS_ao'][0] + p['tS_ao'], p['S_ao'], ax1, ['lightcoral', 'darkred'])
        plot_data(pred['tS_lk'], pred['S_lk'], pred['tS_lk'][0] + p['tS_lk'], p['S_lk'], ax3, ['cornflowerblue', 'darkblue'])
        plot_data(pred['tS_rk'], pred['S_rk'], pred['tS_rk'][0] + p['tS_rk'], p['S_rk'], ax5, ['cornflowerblue', 'darkblue'])

        # Plot concentrations
        v_b_lk = result['pars']['v_p_lk'] / (1 - result['pars']['H'])
        v_b_rk = result['pars']['v_p_rk'] / (1 - result['pars']['H'])

        ax2.set_title('Concentration') 
        ax2.set(xlabel='Time (min)', ylabel='Concentration (mM)', xlim=xlim)
        ax2.plot(pred['tC'] / 60, 0 * pred['tC'], color='gray')
        ax2.plot(pred['tC'] / 60, 1000 * pred['C_ao'][0], linestyle='-', color='darkred', linewidth=2.0, label='Aorta')
        ax2.plot(pred['tC'] / 60, 1000 * pred['C_lk'][0] / v_b_lk, linestyle='--', color='lightcoral', linewidth=2.0, label='Left Kidney')
        ax2.plot(pred['tC'] / 60, 1000 * pred['C_rk'][0] / v_b_rk, linestyle='-.', color='lightcoral', linewidth=2.0, label='Right Kidney')
        ax2.legend()

        def plot_conc_kidney(C, kid, ax):
            ax.set(xlabel='Time (min)', ylabel=f'{kid} conc (mM)', xlim=xlim)
            ax.plot(pred['tC'] / 60, 0 * pred['tC'], color='gray')

            ax.plot(pred['tC'] / 60, 1000 * C.sum(axis=0), linestyle='-', color='darkblue', linewidth=2.0, label='Tissue')
            ax.legend()


        plot_conc_kidney(pred['C_lk'], 'Left kidney', ax4)
        plot_conc_kidney(pred['C_rk'], 'Right kidney', ax6)
